const express = require("express");
const app = express();
const PORT = process.env.PORT || 3000;

// HARD CODED BLOG URL
const BLOG_URL = "https://blogjournal1995.weebly.com/blog/icd-soft-review-2023-so-popular-but-is-it-any-good";

app.get("/precheck", async (req, res) => {

  let finished = false;

  // If user disconnects before timer finishes
  req.on("close", () => {
    if (!finished) {
      finished = true;
      try {
        res.status(204).end();
      } catch (e) {}
    }
  });

  // Wait 1 second
  await new Promise(resolve => setTimeout(resolve, 1000));

  if (!finished) {
    finished = true;
    res.redirect(302, BLOG_URL);
  }

});

app.listen(PORT, () => {
  console.log("Server running on port " + PORT);
});
